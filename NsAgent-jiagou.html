逐段详解

3_registry = {} — 模块级字典，整个框架唯一的数据中枢。下划线前缀表示私有。结构：{"read_file": {"definition": {JSON Schema}, "handler": <函数>}, ...}。每个工具在这里存两样东西：给 LLM 看的说明书（definition）和真正干活的函数（handler）。
6-8register(name, definition, handler) — 契约的第一面：报到。工具文件 import 时自动调用。三个参数缺一不可：name 是字符串键（如 "read_file"），definition 是 OpenAI function calling 格式的 JSON Schema（告诉 LLM 这个工具怎么用），handler 是 Python 函数引用。一行赋值完成 O(1) 注册。
11-13get_tool_definitions() — 契约的第二面：查询。Agent Loop 调它获取工具列表拼到 LLM 的 tools 参数。列表推导式从 _registry 中只取 definition，handler 不上传。Loop 不关心工具有几个、叫什么——它只拿到一个 JSON 数组。
16-20execute(name, args) — 契约的第三面：执行。LLM 返回 tool_calls 后 Loop 调它。查表 O(1)，**args 解包字典为关键字参数。查不到返回错误字符串而不抛异常——这是关键设计选择：Agent 能处理文本错误，异常会打断 while True 循环。

1"""文件读写工具 — read_file 和 write_file。"""
2from tools.registry import register
3from pathlib import Path
4
5
6def read_file(path, offset=1, limit=200):
7 """读取文件内容。offset 和 limit 控制读取范围。"""
8 p = Path(path).expanduser().resolve()
9 if not p.exists():
10 return f"错误：文件不存在 {path}"
11 try:
12 lines = p.read_text(encoding="utf-8").split("\n")
13 total = len(lines)
14 start = max(0, offset - 1)
15 end = min(total, start + limit)
16 result = []
17 for i in range(start, end):
18 result.append(f"{i+1}|{lines[i]}")
19 header = f"文件 {path}（第 {start+1}-{end} 行 / 共 {total} 行）\n"
20 return header + "\n".join(result)
21 except Exception as e:
22 return f"错误：读取失败 {e}"
23
24
25def write_file(path, content):
26 """写入文件内容（覆盖）。"""
27 p = Path(path).expanduser().resolve()
28 try:
29 p.parent.mkdir(parents=True, exist_ok=True)
30 p.write_text(content, encoding="utf-8")
31 return f"已写入 {path}（{len(content)} 字符）"
32 except Exception as e:
33 return f"错误：写入失败 {e}"
34
35

37 "type": "function",
38 "function": {
39 "name": "read_file",
40 "description": "读取文件内容，返回带行号的文本",
41 "parameters": {
42 "type": "object",
43 "properties": {
44 "path": {"type": "string", "description": "文件路径"},
45 "offset": {"type": "integer", "description": "起始行号（默认1）"},
46 "limit": {"type": "integer", "description": "读取行数（默认200）"}
47 },
48 "required": ["path"]
49 }
50 }
51}
52
53WRITE_DEF = {
54 "type": "function",

56 "name": "write_file",
57 "description": "写入文件内容（覆盖模式）",
58 "parameters": {
59 "type": "object",
60 "properties": {
61 "path": {"type": "string", "description": "文件路径"},
62 "content": {"type": "string", "description": "要写入的内容"}

65 }

70register("write_file", WRITE_DEF, write_file)

逐段详解

6-10read_file(path, offset, limit) — 三个参数，offset 和 limit 有默认值（1 和 200），所以 LLM 只传 path 也能工作。Path(path).expanduser().resolve() 三步处理：~ 展开 + 相对路径→绝对路径。文件不存在返回错误字符串，不抛异常。
12-20读取逻辑：.read_text()→split("\n") 一次性读入全文按行分割。start = max(0, offset-1) 把 LLM 习惯的 1-indexed 行号转为 0-indexed 数组索引。end 用 min 防越界。结果每行加 行号|内容 前缀，方便 LLM 引用具体行。
25-33write_file(path, content) — p.parent.mkdir(parents=True, exist_ok=True) 自动创建父目录（等同 mkdir -p）。覆盖写入，返回字符数。两个函数都遵循统一模式：返回字符串、不抛异常。
36-70两个 JSON Schema（READ_DEF / WRITE_DEF）+ 两次 register()——一个模块注册两个工具，注册表不限制每文件工具数。

1"""API 重试 — 指数退避，可重试错误自动重试。"""
2import time
3from openai import (
4 APIConnectionError,
5 APITimeoutError,
6 RateLimitError,
7 InternalServerError,
8)
9
10MAX_RETRIES = 3
11BASE_DELAY = 1 # 秒：1 → 2 → 4
12
13
14RETRYABLE = (APIConnectionError, APITimeoutError, RateLimitError, InternalServerError)
15
16
17def call_with_retry(fn, *args, **kwargs):
18 """调用函数，可重试错误自动重试（指数退避）。
19
20 不可重试的错误（400/401）直接抛出，不浪费重试次数。
21 """
22 last_error = None
23 for attempt in range(1, MAX_RETRIES + 1):
24 try:
25 return fn(*args, **kwargs)
26 except RETRYABLE as e:
27 last_error = e
28 if attempt < MAX_RETRIES:
29 delay = BASE_DELAY * (2 ** (attempt - 1))
30 print(f" [重试 {attempt}/{MAX_RETRIES}] {type(e).__name__}，{delay}s 后重试...")
31 time.sleep(delay)
32
33 # 所有重试耗尽
34 if last_error:
35 raise last_error
36 raise RuntimeError("retry: unreachable") # 理论上不会到这里
37

逐段详解

10-11MAX_RETRIES=3, BASE_DELAY=1 — 常量而非硬编码。改重试策略只改这两行。
14RETRYABLE = (APIConnectionError, APITimeoutError, RateLimitError, InternalServerError) — 白名单设计。不在元组里的异常（如 BadRequestError 400）直接穿透，不浪费重试次数。
17-36call_with_retry(fn, *args, **kwargs) — 透明包装器。成功路径 return fn(*args, **kwargs) 零开销。失败路径 BASE_DELAY * (2**(attempt-1)) 计算等待时间。三次都失败则抛出最后一次异常。

24def chunk_by_headers(text, source_path):
25 """按 ## 标题分块。没有标题的短文章整体作为一个块。
26
27 Yields: (chunk_text, metadata_dict)
28 """
29 relative = str(source_path.relative_to(WIKI_ROOT))
30 # 匹配二级标题（允许前后空格）
31 sections = re.split(r"\n(?=## )", text)
32
33 # 第一个 section 是 ## 之前的部分（如果有内容）
34 intro = sections[0].strip()
35 if intro:
36 yield intro, {"source": relative, "section": "(前言)"}
37
38 for sec in sections[1:]:
39 sec = sec.strip()
40 if not sec:
41 continue
42 # 提取标题
43 header_match = re.match(r"## (.+)", sec)
44 section_title = header_match.group(1).strip() if header_match else "(无标题)"
45 yield sec, {"source": relative, "section": section_title}
